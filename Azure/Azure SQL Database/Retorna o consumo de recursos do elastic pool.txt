SELECT 
    CONVERT(varchar(19), DATEADD(HOUR, -3, start_time), 120) AS start_time_utc_3,
    CONVERT(varchar(19), DATEADD(HOUR, -3, end_time), 120) AS end_time_utc_3,
    elastic_pool_name,
    (
        SELECT MAX(v) 
        FROM (VALUES 
            (avg_cpu_percent), 
            (avg_data_io_percent), 
            (avg_log_write_percent)
        ) AS value(v)
    ) AS [avg_DTU_percent]
FROM sys.elastic_pool_resource_stats
WHERE elastic_pool_name = 'els-aula-sql'
ORDER BY end_time DESC;

-- Média por hora em UTC -3 com data completa
SELECT 
    CONVERT(date, DATEADD(HOUR, -3, end_time)) AS Data_UTC_3,
    DATEPART(HOUR, DATEADD(HOUR, -3, end_time)) AS Hora_UTC_3,
    AVG(avg_cpu_percent) AS AVG_CPU, 
    AVG(avg_data_io_percent) AS AVG_IO_Dados,
    AVG(avg_log_write_percent) AS AVG_Log
FROM sys.elastic_pool_resource_stats
WHERE elastic_pool_name = 'els-aula-sql-test'
GROUP BY 
    CONVERT(date, DATEADD(HOUR, -3, end_time)),
    DATEPART(HOUR, DATEADD(HOUR, -3, end_time))
ORDER BY 
    Data_UTC_3, 
    Hora_UTC_3;
