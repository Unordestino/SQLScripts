--Geralmente usada para verificar se o job está executando na primária ou secundária, se for secundária será apresentado um erro e  job não prossegue

CREATE   PROCEDURE [dbo].[usp_CheckIsPrimary]
  @DatabaseName SYSNAME = NULL  -- # NULL = verifica qualquer AG local
AS BEGIN
  SET NOCOUNT ON;

  IF (@DatabaseName IS NOT NULL) BEGIN
    IF (sys.fn_hadr_is_primary_replica(@DatabaseName) = 0)
      THROW 50001, 'Réplica não é primária.', 1;
    RETURN;
  END

  IF (NOT EXISTS (
    SELECT 1
    FROM        [sys].[dm_hadr_availability_replica_states]   AS rs
    INNER JOIN  [sys].[availability_replicas]                 AS ar
      ON  (1 = 1)
      AND (rs.replica_id = ar.replica_id)
    WHERE (1 = 1)
      AND (rs.is_local = 1)
      AND (rs.role_desc = N'PRIMARY')
    ))
    THROW 50001, 'Réplica não é primária.', 1;
END;
GO
